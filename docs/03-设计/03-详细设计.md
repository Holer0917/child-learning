## 1. Core 层详细设计

### 1.1 事件总线 `core/bus.js`

**职责**：极简的发布订阅，用于解耦 UI 与 Domain。

```js
/**
 * 创建事件总线
 * @returns {{on:Function, off:Function, emit:Function, once:Function}}
 */
export function createBus() {
  /** @type {Map<string, Set<Function>>} */
  const handlers = new Map();

  function on(type, fn) {
    if (!handlers.has(type)) handlers.set(type, new Set());
    handlers.get(type).add(fn);
    return () => off(type, fn);   // 返回取消订阅函数
  }

  function off(type, fn) {
    handlers.get(type)?.delete(fn);
  }

  function once(type, fn) {
    const wrap = (payload) => { off(type, wrap); fn(payload); };
    return on(type, wrap);
  }

  function emit(type, payload) {
    const set = handlers.get(type);
    if (!set) return;
    // 复制一份，避免回调中取消订阅导致的迭代异常
    for (const fn of [...set]) {
      try { fn(payload); }
      catch (err) { console.error(`[bus] handler error on "${type}"`, err); }
    }
  }

  return { on, off, once, emit };
}

export const bus = createBus();
```

**设计要点**
- 单个 handler 抛错不影响其它 handler（try/catch 隔离）；
- `emit` 遍历副本，避免回调中修改订阅集合导致跳过；
- `on` 返回取消订阅函数，便于组件卸载时清理。

### 1.2 持久化 `core/storage.js`

```js
const PREFIX = 'km:';
const META_KEY = PREFIX + 'meta';
const CURRENT_VERSION = 1;

/** 检测 localStorage 是否可用（隐私模式下会抛错） */
export function isAvailable() {
  try {
    const k = '__km_test__';
    localStorage.setItem(k, '1');
    localStorage.removeItem(k);
    return true;
  } catch { return false; }
}

/** 内存兜底：localStorage 不可用时退化为会话内存储 */
const memoryFallback = new Map();

export function read(key, fallback = null) {
  try {
    const raw = isAvailable()
      ? localStorage.getItem(PREFIX + key)
      : memoryFallback.get(key);
    if (raw == null) return fallback;
    return JSON.parse(raw);
  } catch (err) {
    // JSON 解析失败 → 数据损坏
    handleCorruption(key, err);
    return fallback;
  }
}

export function write(key, value) {
  const raw = JSON.stringify(value);
  try {
    if (isAvailable()) localStorage.setItem(PREFIX + key, raw);
    else memoryFallback.set(key, raw);
    bus.emit('data:changed', { keys: [key] });
    return true;
  } catch (err) {
    // QuotaExceededError
    bus.emit('storage:error', { reason: 'quota', key });
    return false;
  }
}

export function remove(key) {
  if (isAvailable()) localStorage.removeItem(PREFIX + key);
  else memoryFallback.delete(key);
}

/** 数据损坏处理：备份损坏数据后以全新状态启动（对应 NFR-03） */
function handleCorruption(key, err) {
  const raw = isAvailable() ? localStorage.getItem(PREFIX + key) : null;
  if (raw) {
    try { localStorage.setItem(PREFIX + '__corrupt__' + key, raw); } catch {}
    localStorage.removeItem(PREFIX + key);
  }
  bus.emit('storage:corrupt', { key });
}
```

### 1.3 路由 `core/router.js`

```js
/**
 * 哈希路由。路由表项：{ path: '#/home', page: renderHome }
 * 支持查询参数：#/mistakes?kp=G3-01
 */
export function createRouter(routes) {
  let currentRoute = null;

  function parse() {
    const hash = location.hash || '#/home';
    const [path, queryStr] = hash.split('?');
    const query = Object.fromEntries(new URLSearchParams(queryStr || ''));
    return { path, query };
  }

  function resolve() {
    const { path, query } = parse();
    const route = routes.find(r => r.path === path) ?? routes[0]; // 未匹配 → 首页
    if (currentRoute?.page?.destroy) currentRoute.page.destroy();
    currentRoute = { ...route, query };
    bus.emit('route:changed', currentRoute);
    return currentRoute;
  }

  function navigate(path, query) {
    const qs = query ? '?' + new URLSearchParams(query) : '';
    location.hash = path + qs;
  }

  window.addEventListener('hashchange', resolve);
  return { resolve, navigate, current: () => currentRoute };
}
```

### 2.2 出题引擎 `domain/generator.js`

```js
import { SYLLABUS, getKnowledgePoint } from './syllabus.js';
import { createRandom } from '../utils/random.js';

/** 生成一份卷子 */
export function generatePaper({ kpIds, count, difficulty = 'normal',
                                types = ['horizontal'], seed } = {}) {
  const rng = createRandom(seed ?? Date.now());
  const questions = [];
  const seen = new Set();

  // 1. 按知识点数量均匀分配题量（余数补给前面的知识点）
  const per = distribute(count, kpIds.length);

  kpIds.forEach((kpId, i) => {
    const kp = getKnowledgePoint(kpId);
    if (!kp) return;
    const usableTypes = types.filter(t => kp.types.includes(t));
    const list = usableTypes.length ? usableTypes : kp.types;

    let made = 0, guard = 0;
    while (made < per[i] && guard < per[i] * 30) {
      guard++;
      const spec = adjustSpec(kp.spec, difficulty);
      const q = generateFromSpec(spec, rng);
      if (!q) continue;

      q.kpId = kpId;
      q.difficulty = kp.difficulty;
      q.type = rng.pick(list);
      q.text = renderText(q, q.type);
      q.id = `q${questions.length + 1}`;

      const key = dedupeKey(q);
      if (seen.has(key)) continue;
      seen.add(key);

      questions.push(q);
      made++;
    }
  });

  rng.shuffle(questions);                      // 打乱顺序，避免同知识点扎堆
  questions.forEach((q, i) => { q.id = `q${i + 1}`; });

  return {
    id: `p${Date.now()}`,
    questions,
    options: { kpIds, count, difficulty, types },
    createdAt: Date.now()
  };
}

/** 把 count 均匀分配到 n 份 */
function distribute(count, n) {
  const base = Math.floor(count / n);
  const rest = count % n;
  return Array.from({ length: n }, (_, i) => base + (i < rest ? 1 : 0));
}
```

**`renderText` — 题干渲染**

```js
/**
 * 依据题型渲染题干文本。
 * horizontal: "156 × 4 = ___"
 * vertical:   结构化数据（由 UI 渲染为竖式）
 * compare:    "156 × 4  ○  600"
 * fill:       "□ × 4 = 624"
 * chain:      "12 + 8 - 5 = ___"
 */
export function renderText(q, type) {
  const OP = { add: '+', sub: '−', mul: '×', div: '÷' };
  switch (type) {
    case 'horizontal':
      return `${q.a} ${OP[q.op]} ${q.b} = ___`;
    case 'vertical':
      return `${q.a} ${OP[q.op]} ${q.b}`;          // 布局信息另存于 layout
    case 'compare':
      return `${q.a} ${OP[q.op]} ${q.b}  ○  ${q.answer}`;
    case 'fill': {
      const hide = q.params.hide;                   // 'a' | 'b' | 'result'
      if (hide === 'b') return `${q.a} ${OP[q.op]} □ = ${q.answer}`;
      if (hide === 'result') return `${q.a} ${OP[q.op]} ${q.b} = ___`;
      return `□ ${OP[q.op]} ${q.b} = ${q.answer}`;
    }
    case 'chain':
      return `${q.params.a} ${OP[q.params.op1]} ${q.params.b} ` +
             `${OP[q.params.op2]} ${q.params.c} = ___`;
    default:
      return `${q.a} ${OP[q.op]} ${q.b} = ___`;
  }
}
```

**竖式版式数据**

```js
/**
 * 为竖式题生成版式数据，供 UI 渲染为对齐的数字网格。
 * @returns {{rows: string[], operator: string, width: number}}
 */
export function buildVerticalLayout(q) {
  const OP = { add: '+', sub: '−', mul: '×', div: '÷' };
  const rows = [String(q.a), OP[q.op] + ' ' + String(q.b)];
  const width = Math.max(...rows.map(r => r.length));
  return {
    rows: rows.map(r => r.padStart(width, ' ')),
    operator: OP[q.op],
    width,
    answer: q.answer
  };
}
```

```js
import { read, write } from '../core/storage.js';

const key = (pid) => `data:${pid}`;

export function readData(pid) {
  return read(key(pid), { sessions: [], mistakes: [], badges: [] });
}

export function writeData(pid, data) {
  return write(key(pid), data);
}

/** 归档错题（含去重与次数累加） */
export function archiveMistake(pid, question, userAnswer) {
  const data = readData(pid);
  const k = dedupeKey(question);
  const exist = data.mistakes.find(m => m.key === k);

  if (exist) {
    exist.wrongCount += 1;
    exist.userAnswer = userAnswer;
    exist.mastered = false;
    exist.lastWrongAt = Date.now();
  } else {
    data.mistakes.push({
      id: `m${Date.now()}${data.mistakes.length}`,
      key: k, kpId: question.kpId, type: question.type,
      text: question.text, answer: question.answer, userAnswer,
      params: question.params, kind: question.kind ?? 'int',
      wrongCount: 1, mastered: false,
      firstWrongAt: Date.now(), lastWrongAt: Date.now()
    });
  }
  writeData(pid, data);
  return exist ?? data.mistakes.at(-1);
}

/** 列出错题，支持筛选 */
export function listMistakes(pid, { kpId, status } = {}) {
  const { mistakes } = readData(pid);
  return mistakes.filter(m =>
    (!kpId || m.kpId === kpId) &&
    (!status || (status === 'pending' ? !m.mastered : m.mastered))
  );
}

/** 按知识点分组 */
export function groupByKP(pid) {
  const map = new Map();
  for (const m of listMistakes(pid, { status: 'pending' })) {
    if (!map.has(m.kpId)) map.set(m.kpId, []);
    map.get(m.kpId).push(m);
  }
  return [...map.entries()].map(([kpId, list]) => ({
    kpId, name: getKnowledgePoint(kpId)?.name ?? kpId,
    count: list.length, list
  }));
}

export function markMastered(pid, mistakeId) {
  const data = readData(pid);
  const m = data.mistakes.find(x => x.id === mistakeId);
  if (m) { m.mastered = true; writeData(pid, data); bus.emit('mistake:mastered', { mistakeId }); }
}

export function clearMastered(pid) {
  const data = readData(pid);
  const before = data.mistakes.length;
  data.mistakes = data.mistakes.filter(m => !m.mastered);
  writeData(pid, data);
  return before - data.mistakes.length;
}

export function countMistakes(pid) {
  const { mistakes } = readData(pid);
  const mastered = mistakes.filter(m => m.mastered).length;
  return { pending: mistakes.length - mastered, mastered, total: mistakes.length };
}
```

### 2.7 计时服务 `domain/timer.js`

```js
/**
 * 练习计时器。支持正计时与倒计时两种模式。
 * @returns {{start, pause, resume, stop, elapsed, remaining, onTick}}
 */
export function createTimer({ mode = 'up', limitMs = 0, onTick, onExpire } = {}) {
  let startedAt = null, accumulated = 0, paused = false, tickId = null;

  function now() {
    return accumulated + (paused || startedAt == null ? 0 : Date.now() - startedAt);
  }

  function tick() {
    const e = now();
    onTick?.(e, mode === 'down' ? Math.max(0, limitMs - e) : e);
    if (mode === 'down' && e >= limitMs) { stop(); onExpire?.(); }
  }

  function start() {
    if (startedAt != null && !paused) return;
    if (paused) { paused = false; startedAt = Date.now(); }
    else { startedAt = Date.now(); accumulated = 0; }
    tickId = setInterval(tick, 250);
  }

  function pause() {
    if (paused || startedAt == null) return;
    accumulated += Date.now() - startedAt;
    paused = true;
    clearInterval(tickId);
  }

  function stop() { pause(); clearInterval(tickId); tickId = null; }

  return { start, pause, resume: start, stop,
           elapsed: now,
           remaining: () => Math.max(0, limitMs - now()) };
}
```

> **计时精度设计**：用 `Date.now()` 差值而非累加 `setInterval` 次数，
> 避免标签页切到后台时浏览器节流导致计时偏慢。

**关键实现**

```js
export function renderQuiz(root, { paper, profile }) {
  const state = {
    index: 0,
    input: '',
    attempts: {},
    phase: 'ANSWERING',
    questionStartedAt: Date.now()
  };

  const timer = createTimer({
    mode: paper.options.limitMs ? 'down' : 'up',
    limitMs: paper.options.limitMs ?? 0,
    onTick: (elapsed) => updateTimerDisplay(elapsed),
    onExpire: () => finish()
  });

  function submit() {
    if (state.phase !== 'ANSWERING') return;
    const q = paper.questions[state.index];
    const result = judgeAnswer(q, state.input);
    state.phase = result.correct ? 'CORRECT' : 'WRONG';

    state.attempts[q.id] = {
      userAnswer: state.input,
      correct: result.correct,
      skipped: result.skipped,
      elapsedMs: Date.now() - state.questionStartedAt
    };

    bus.emit('quiz:answered', { questionId: q.id, correct: result.correct });
    renderFeedback(result, q);

    if (result.correct) {
      setTimeout(next, 400);                 // 答对自动进入下一题
    }
    // 答错则停留在 WRONG 态，等待用户点击
  }

  function next() {
    state.index++;
    state.input = '';
    state.phase = 'ANSWERING';
    state.questionStartedAt = Date.now();
    if (state.index >= paper.questions.length) return finish();
    renderQuestion();
  }

  function skip() {
    const q = paper.questions[state.index];
    state.attempts[q.id] = { userAnswer: '', correct: false,
                             skipped: true, elapsedMs: 0 };
    next();
  }

  function finish() {
    timer.stop();
    const result = settleSession(paper, state.attempts, timer.elapsed());
    bus.emit('quiz:finished', { paper, result });

    // 归档错题
    for (const { question, userAnswer } of result.wrongList) {
      archiveMistake(profile.id, question, userAnswer);
    }
    // 写入历史
    appendSession(profile.id, { ...result, paper });

    navigate('#/result', { sid: result.id });
  }

  timer.start();
  renderQuestion();
  return { destroy: () => timer.stop() };
}
```

### 3.2 竖式题渲染 `ui/components/question.js`

```js
/**
 * 渲染竖式题目为对齐的数字网格。
 * 使用 CSS Grid，每个字符一个单元格，保证数位对齐。
 *
 *    ┌─────────────┐
 *    │    1 5 6    │
 *    │  ×     4    │
 *    │  ─────────  │
 *    │    □ □ □    │
 *    └─────────────┘
 */
export function renderVertical(q) {
  const layout = buildVerticalLayout(q);
  const digits = String(q.a).length + 2;      // 预留运算符列

  const el = document.createElement('div');
  el.className = 'vertical';
  el.style.setProperty('--cols', digits);

  // 被乘数/被除数行
  el.appendChild(verticalRow(String(q.a), digits, 'right'));
  // 运算符 + 操作数行
  el.appendChild(verticalRow(OP[q.op] + String(q.b), digits, 'right'));
  // 分隔线
  el.appendChild(Object.assign(document.createElement('div'),
    { className: 'vertical-line' }));
  // 作答行（空白框）
  el.appendChild(verticalAnswerRow(digits));

  el.setAttribute('role', 'math');
  el.setAttribute('aria-label', `${q.a} ${OP[q.op]} ${q.b}，请用竖式计算`);
  return el;
}
```

### 3.3 图表 `ui/charts/line.js`

```js
/**
 * 折线图（正确率趋势）。纯 SVG，无依赖。
 * 配色遵循无障碍规范：单序列使用 --series-1，网格线使用 --grid。
 */
export function renderLineChart(container, data, { yMax = 100, yMin = 0 } = {}) {
  const W = container.clientWidth || 560;
  const H = 200;
  const PAD = { top: 16, right: 16, bottom: 28, left: 40 };
  const iw = W - PAD.left - PAD.right;
  const ih = H - PAD.top - PAD.bottom;

  const x = (i) => PAD.left + (data.length === 1 ? iw / 2
                          : (i / (data.length - 1)) * iw);
  const y = (v) => PAD.top + ih - ((v - yMin) / (yMax - yMin)) * ih;

  const svg = svgEl('svg', { viewBox: `0 0 ${W} ${H}`, class: 'chart' });

  // 网格线（3 条水平参考线）
  for (const v of [yMin, (yMin + yMax) / 2, yMax]) {
    svg.appendChild(svgEl('line', {
      x1: PAD.left, x2: W - PAD.right, y1: y(v), y2: y(v),
      class: 'chart-grid'
    }));
    svg.appendChild(svgEl('text', {
      x: PAD.left - 6, y: y(v) + 4, class: 'chart-tick', 'text-anchor': 'end'
    }, `${v}%`));
  }

  // 折线
  const d = data.map((p, i) => `${i ? 'L' : 'M'}${x(i)},${y(p.value)}`).join(' ');
  svg.appendChild(svgEl('path', { d, class: 'chart-line' }));

  // 数据点 + 末点直接标注（选择性直接标签）
  data.forEach((p, i) => {
    svg.appendChild(svgEl('circle', { cx: x(i), cy: y(p.value), r: 3.5,
                                      class: 'chart-dot' }));
  });
  const last = data.at(-1);
  svg.appendChild(svgEl('text', {
    x: x(data.length - 1), y: y(last.value) - 10,
    class: 'chart-label', 'text-anchor': 'middle'
  }, `${last.value}%`));

  // X 轴标签（最多 7 个，避免重叠）
  const step = Math.max(1, Math.ceil(data.length / 7));
  data.forEach((p, i) => {
    if (i % step !== 0 && i !== data.length - 1) return;
    svg.appendChild(svgEl('text', {
      x: x(i), y: H - 8, class: 'chart-tick', 'text-anchor': 'middle'
    }, p.label));
  });

  container.replaceChildren(svg);
}
```

**无障碍**
- 图表容器提供 `role="img"` 与 `aria-label`（如"正确率趋势图，最近 7 次，从 72% 上升到 86.7%"）；
- 同时提供"查看数据表"的折叠表格（对应 dataviz 规范的 table view 要求）；
- 网格线与坐标轴使用弱化的 `--grid` / `--muted` 色，不与数据序列争夺注意力。

## 5. 应用入口 `src/main.js`

```js
import { bus } from './core/bus.js';
import { createRouter } from './core/router.js';
import { migrate } from './core/storage.js';
import { renderShell } from './ui/shell.js';
import * as pages from './ui/pages/index.js';

// 1. 数据结构迁移
migrate();

// 2. 初始化外壳
const app = document.getElementById('app');
renderShell(app);

// 3. 注册路由
const outlet = app.querySelector('#outlet');
const ROUTES = [
  { path: '#/home',     page: pages.home },
  { path: '#/syllabus', page: pages.syllabus },
  { path: '#/paper',    page: pages.paper },
  { path: '#/quiz',     page: pages.quiz },
  { path: '#/result',   page: pages.result },
  { path: '#/mistakes', page: pages.mistakes },
  { path: '#/variants', page: pages.variants },
  { path: '#/stats',    page: pages.stats },
  { path: '#/history',  page: pages.history },
  { path: '#/profiles', page: pages.profiles },
  { path: '#/settings', page: pages.settings },
  { path: '#/print',    page: pages.print }
];

const router = createRouter(ROUTES);
bus.on('route:changed', (route) => {
  outlet.replaceChildren();
  route.page.render(outlet, route.query);
});
router.resolve();

// 4. 全局错误兜底
window.addEventListener('error', (e) => {
  console.error('[global]', e.error);
  showToast('出了点小问题，请重试');
});
bus.on('storage:error', ({ reason }) => {
  showToast(reason === 'quota'
    ? '存储空间不足，请先导出数据再清理旧记录'
    : '数据保存失败，请检查浏览器设置');
});
bus.on('storage:corrupt', () => {
  showToast('本地数据损坏，已重置为初始状态');
});

// 5. 注册 Service Worker（离线可用）
if ('serviceWorker' in navigator && location.protocol === 'https:') {
  navigator.serviceWorker.register('./sw.js').catch(() => {});
}
```
